Basic Invoice Registration

Complete Node.js Example

const https = require('https');
const fs = require('fs');

class AEATClient {
  constructor(config) {
    this.httpConfig = {
      hostname: config.hostname ?? 'prewww1.aeat.es',
      port: 443,
      path: '/wlpl/TIKE-CONT/ws/SistemaFacturacion/VerifactuSOAP',
      ...config
    };

    this.certificate = {
      key: fs.readFileSync(config.keyPath),
      cert: fs.readFileSync(config.certPath),
      ca: fs.existsSync(config.caPath) ? fs.readFileSync(config.caPath) : undefined
    };
  }

  async registerInvoice(invoice) {
    const soapBody = this.createRegisterRequest(invoice);
    const xml = await this.makeRequest(soapBody);
    return this.parseAltaResponse(xml);
  }

  async cancelInvoice(invoice) {
    const soapBody = this.createCancelRequest(invoice);
    const xml = await this.makeRequest(soapBody);
    return this.parseBajaResponse(xml);
  }

  async queryInvoice(filters) {
    const soapBody = this.createQueryRequest(filters);
    const xml = await this.makeRequest(soapBody);
    return this.parseConsultaResponse(xml);
  }

  createRegisterRequest(invoice) {
    const waitTime = invoice.tiempoEsperaEnvio ?? 60;
    return `<?xml version="1.0" encoding="UTF-8"?>
<soapenv:Envelope xmlns:soapenv="http://schemas.xmlsoap.org/soap/envelope/"
                  xmlns:ver="urn:aeat:verifactu:schema">
  <soapenv:Header/>
  <soapenv:Body>
    <ver:MensajeAlta>
      <ver:Cabecera>
        <ver:NIFEmisor>${invoice.nifEmisor}</ver:NIFEmisor>
        <ver:FechaEnvio>${invoice.fechaEnvio}</ver:FechaEnvio>
        <ver:TiempoEsperaEnvio>${waitTime}</ver:TiempoEsperaEnvio>
        <ver:TipoComunicacion>Alta</ver:TipoComunicacion>
      </ver:Cabecera>
      <ver:Registros>
        <ver:Registro>
          <ver:Serie>${invoice.serie}</ver:Serie>
          <ver:NumFactura>${invoice.numero}</ver:NumFactura>
          <ver:FechaExpedicion>${invoice.fechaExpedicion}</ver:FechaExpedicion>
          <ver:ImporteTotal>${invoice.importeTotal}</ver:ImporteTotal>
          <!-- add DetalleDesglose, Encadenamiento, SistemaInformatico, Firma, etc. -->
        </ver:Registro>
      </ver:Registros>
    </ver:MensajeAlta>
  </soapenv:Body>
</soapenv:Envelope>`;
  }

  createCancelRequest(invoice) {
    const waitTime = invoice.tiempoEsperaEnvio ?? 60;
    return `<?xml version="1.0" encoding="UTF-8"?>
<soapenv:Envelope xmlns:soapenv="http://schemas.xmlsoap.org/soap/envelope/"
                  xmlns:ver="urn:aeat:verifactu:schema">
  <soapenv:Header/>
  <soapenv:Body>
    <ver:MensajeBaja>
      <ver:Cabecera>
        <ver:NIFEmisor>${invoice.nifEmisor}</ver:NIFEmisor>
        <ver:FechaEnvio>${invoice.fechaEnvio}</ver:FechaEnvio>
        <ver:TiempoEsperaEnvio>${waitTime}</ver:TiempoEsperaEnvio>
        <ver:TipoComunicacion>Baja</ver:TipoComunicacion>
      </ver:Cabecera>
      <ver:Registros>
        <ver:Registro>
          <ver:Serie>${invoice.serie}</ver:Serie>
          <ver:NumFactura>${invoice.numero}</ver:NumFactura>
          <ver:Motivo>${invoice.motivo}</ver:Motivo>
        </ver:Registro>
      </ver:Registros>
    </ver:MensajeBaja>
  </soapenv:Body>
</soapenv:Envelope>`;
  }

  createQueryRequest(filters) {
    const waitTime = filters.tiempoEsperaEnvio ?? 60;
    return `<?xml version="1.0" encoding="UTF-8"?>
<soapenv:Envelope xmlns:soapenv="http://schemas.xmlsoap.org/soap/envelope/"
                  xmlns:ver="urn:aeat:verifactu:schema">
  <soapenv:Header/>
  <soapenv:Body>
    <ver:MensajeConsulta>
      <ver:Cabecera>
        <ver:NIFEmisor>${filters.nifEmisor}</ver:NIFEmisor>
        <ver:FechaEnvio>${filters.fechaEnvio}</ver:FechaEnvio>
        <ver:TiempoEsperaEnvio>${waitTime}</ver:TiempoEsperaEnvio>
        <ver:TipoComunicacion>Consulta</ver:TipoComunicacion>
      </ver:Cabecera>
      ${filters.clavePaginacion ? `<ver:Paginacion>
        <ver:ClavePaginacion>${filters.clavePaginacion}</ver:ClavePaginacion>
      </ver:Paginacion>` : `<ver:Filtros>
        <ver:RangoFechaDesde>${filters.fechaDesde}</ver:RangoFechaDesde>
        <ver:RangoFechaHasta>${filters.fechaHasta}</ver:RangoFechaHasta>
      </ver:Filtros>`}
    </ver:MensajeConsulta>
  </soapenv:Body>
</soapenv:Envelope>`;
  }

  makeRequest(soapBody) {
    return new Promise((resolve, reject) => {
      const options = {
        ...this.httpConfig,
        ...this.certificate,
        method: 'POST',
        headers: {
          'Content-Type': 'text/xml; charset=utf-8',
          'SOAPAction': ''
        }
      };

      const req = https.request(options, (res) => {
        let data = '';
        res.on('data', chunk => { data += chunk; });
        res.on('end', () => resolve(data));
      });

      req.on('error', reject);
      req.write(soapBody);
      req.end();
    });
  }

  parseAltaResponse(xml) {
    const estado = this.capture(xml, /(.*?)<\/ver:EstadoEnvio>/);
    const csv = this.capture(xml, /(.*?)<\/ver:CSV>/);
    const waitTime = this.capture(xml, /(.*?)<\/ver:TiempoEsperaEnvio>/);
    return { estadoEnvio: estado, csv, tiempoEsperaEnvio: waitTime ? Number(waitTime) : null };
  }

  parseBajaResponse(xml) {
    const estado = this.capture(xml, /(.*?)<\/ver:EstadoEnvio>/);
    const waitTime = this.capture(xml, /(.*?)<\/ver:TiempoEsperaEnvio>/);
    return { estadoEnvio: estado, tiempoEsperaEnvio: waitTime ? Number(waitTime) : null };
  }

  parseConsultaResponse(xml) {
    const estado = this.capture(xml, /(.*?)<\/ver:EstadoEnvio>/);
    const indicador = this.capture(xml, /(.*?)<\/ver:IndicadorPaginacion>/);
    const clave = this.capture(xml, /(.*?)<\/ver:ClavePaginacion>/);
    const csvs = [...xml.matchAll(/(.*?)<\/ver:CSV>/g)].map(m => m[1]);
    return {
      estadoEnvio: estado,
      indicadorPaginacion: indicador,
      clavePaginacion: clave || null,
      csv: csvs
    };
  }

  capture(xml, regex) {
    const match = xml.match(regex);
    return match ? match[1] : null;
  }
}

Database Integration

MySQL Integration Example

const mysql = require('mysql2/promise');

class InvoiceService {
  constructor(dbConfig) {
    this.db = mysql.createConnection(dbConfig);
  }
  
  async processInvoice(invoiceId) {
    try {
      // Get invoice from database
      const invoice = await this.getInvoice(invoiceId);
      
      // Register with AEAT
      const aeatResult = await this.registerWithAEAT(invoice);
      
      // Update database with result
      await this.updateInvoiceStatus(invoiceId, aeatResult);
      
      return aeatResult;
    } catch (error) {
      // Log error and update database
      await this.logError(invoiceId, error.message);
      throw error;
    }
  }
  
  async getInvoice(invoiceId) {
    const [rows] = await this.db.execute(
      'SELECT * FROM invoices WHERE id = ?',
      [invoiceId]
    );
    
    if (rows.length === 0) {
      throw new Error('Invoice not found');
    }
    
    return rows[0];
  }
  
  async updateInvoiceStatus(invoiceId, result) {
    const query = `
      UPDATE invoices 
      SET aeat_csv_code = ?, 
          aeat_status = ?, 
          aeat_error = NULL,
          updated_at = NOW()
      WHERE id = ?
    `;
    
    const status = result.success ? 'verified' : 'error';
    await this.db.execute(query, [result.csv, status, invoiceId]);
  }
  
  async logError(invoiceId, errorMessage) {
    const query = `
      UPDATE invoices 
      SET aeat_status = 'error', 
          aeat_error = ?,
          updated_at = NOW()
      WHERE id = ?
    `;
    
    await this.db.execute(query, [errorMessage, invoiceId]);
  }
}

Error Handling Examples

Comprehensive Error Handling

class AEATService {
  constructor(config) {
    this.config = config;
    this.retryAttempts = 3;
    this.retryDelay = 1000;
  }
  
  async registerInvoiceWithRetry(invoice) {
    let lastError;
    
    for (let attempt = 1; attempt <= this.retryAttempts; attempt++) {
      try {
        return await this.registerInvoice(invoice);
      } catch (error) {
        lastError = error;
        
        if (this.isRetryableError(error)) {
          console.log(`Attempt ${attempt} failed, retrying in ${this.retryDelay}ms...`);
          await this.delay(this.retryDelay * attempt);
          continue;
        } else {
          throw error;
        }
      }
    }
    
    throw new Error(`All ${this.retryAttempts} attempts failed. Last error: ${lastError.message}`);
  }
  
  isRetryableError(error) {
    const retryableErrors = [
      'ECONNRESET',
      'ETIMEDOUT',
      'ENOTFOUND',
      'ECONNREFUSED'
    ];
    
    return retryableErrors.some(code => error.code === code);
  }
  
  delay(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

const assert = require('assert');
const AEATClient = require('./aeat-client');

describe('AEAT Client', () => {
  let client;

    client = new AEATClient({
      keyPath: './test/certificates/client-key.pem',
      certPath: './test/certificates/client-cert.pem',

      numero: '00001',
      importeTotal: '100.00',
